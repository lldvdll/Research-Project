% Custom Commands
\newcommand{\singlefigure}{0.45\textwidth}

% Width of ONE panel of a Mode A group composed with subfigure. The 900 scripts save at final
% physical size (src/style.py), so every \includegraphics below is included at its natural width
% with NO scale factor -- scaling an image scales its fonts with it, and an 8pt axis label
% included at 0.9 lands as 7.2pt. If a figure does not fit, change the figsize in the 900 script
% and re-run it; do not scale it here.
\newlength{\panelwidth}
\setlength{\panelwidth}{0.48\textwidth}

% \fig{901_name.png} -- include the 900-series figure if it has been generated, otherwise draw a
% labelled placeholder. Without this the skeleton does not compile until every one of the 18
% main-text figures exists, which would mean no readable draft for weeks. \IfFileExists does not
% consult \graphicspath, so the path is spelled out.
% \detokenize is required: every 900 filename contains underscores, which are subscript
% operators in text mode, so a bare \texttt{#1} fails to compile on exactly the figures that do
% not exist yet.
\newcommand{\fig}[1]{%
  \IfFileExists{../experiments/#1}%
    {\includegraphics{#1}}%
    {\fbox{\parbox[c][0.16\textheight][c]{0.92\linewidth}{\centering
       \textcolor{orange}{\texttt{\detokenize{#1}} --- not generated yet}}}}%
}
% \sim is math-only and < > are not literal in text mode, so the original form of this macro
% failed to compile the moment anything used it.
\newcommand{\wordcount}[1]{\textcolor{red}{\textbf{[write ${\sim}$#1 words]}}}
\newcommand{\gap}[1]{\textit{\textcolor{orange}{[gap: #1]}}}

% FIGURE GRIDS  (first-draft layout, 2026-09-11)
%
% House style: two-column body, with each TOPIC's figures collected into one
% grid spanning the full page width, ruled off in a box, panels lettered
% A, B, C by the subcaption package rather than by hand.
%
%   \begin{figure*}[t]\centering
%     \begin{panelbox}
%       \panel{0.49\panelspan}{a.png}\hfill
%       \panel{0.49\panelspan}{b.png}   % A and B side by side
%       \par\vspace{6pt}                % next row
%       \solo{single.png}                 % unlettered, one-panel grid
%     \end{panelbox}
%     \caption{...}                       % sits below the rule
%   \end{figure*}
%
% WARNING -- \gridfig SCALES, which contradicts the "no scale factor" note
% further up this preamble. That note assumed the 900 scripts save at final
% physical size; measured on 2026-09-11 they do not. Natural widths run from
% 6.9 in to 21.2 in against a 7.1 in page, which is exactly why figures ran
% off the page. Scaling to the panel width is a DRAFT compromise that makes
% the layout readable now. The real fix is to set figsize in the over-wide
% scripts and re-run them, after which the width option can come back out.
% Anything worse than about 2x over-wide still has axis labels too small to
% read comfortably; those are listed in claude_code_management/.
% =========================================================================

% Panel letters: bold "A", no parentheses, above-left of the panel.
\DeclareCaptionLabelFormat{panelletter}{\textbf{#2}}
\captionsetup[subfigure]{labelformat=panelletter, labelsep=none,
                         justification=raggedright, singlelinecheck=false,
                         skip=1pt, position=top}
\renewcommand{\thesubfigure}{\Alph{subfigure}}

% Width available inside the box, once \fboxsep and the rule are taken off.
\newlength{\panelspan}
\setlength{\panelspan}{\textwidth}

% One figure scaled to the width of its panel, with the same not-generated
% fallback \fig uses.
\newcommand{\gridfig}[1]{%
  \IfFileExists{../experiments/#1}%
    {\includegraphics[width=\linewidth]{#1}}%
    {\fbox{\parbox[c][0.13\textheight][c]{0.94\linewidth}{\centering
       \textcolor{orange}{\texttt{\detokenize{#1}} --- not generated yet}}}}%
}

% A lettered panel: #1 = width, #2 = file.
\newcommand{\panel}[2]{%
  \begin{subfigure}[t]{#1}\subcaption{}\gridfig{#2}\end{subfigure}%
}

% An unlettered panel, for a grid holding only one figure.
\newcommand{\solo}[1]{\gridfig{#1}}



\newsavebox{\gridbox}

% Rules a box around a group of panels. Used inside figure*, with \caption
% placed after it so the caption sits below the rule rather than inside it.
\newenvironment{panelbox}%
  {\setlength{\fboxsep}{5pt}%
   \setlength{\panelspan}{\dimexpr\linewidth-2\fboxsep-2\fboxrule\relax}%
   \begin{lrbox}{\gridbox}%
   \begin{minipage}{\panelspan}\centering}%
  {\end{minipage}\end{lrbox}%
   \fbox{\usebox{\gridbox}}}
